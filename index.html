const ROOT_RETROALIMENTACION = "1fHWBgM1j8cKPY5z248dTNfGpMHH3U6Dw"; function limpiar(txt) { if (!txt) return ""; return txt.normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().replace(/[\s_]/g, ""); } function doGet(e) { const action = e.parameter.action; try { const rootFolder = DriveApp.getFolderById(ROOT_RETROALIMENTACION); if (action === "semestres") { const folders = rootFolder.getFolders(); let lista = []; while (folders.hasNext()) lista.push(folders.next().getName()); return ContentService.createTextOutput(JSON.stringify(lista)).setMimeType(ContentService.MimeType.JSON); } if (action === "grupos") { const semestreFolder = getFolderByName(rootFolder, e.parameter.semestre); if (!semestreFolder) return errorJSON("Semestre no encontrado"); const folders = semestreFolder.getFolders(); let lista = []; while (folders.hasNext()) lista.push(folders.next().getName()); return ContentService.createTextOutput(JSON.stringify(lista)).setMimeType(ContentService.MimeType.JSON); } if (action === "estudiantes") { const grupoFolder = getGrupoFolder(rootFolder, e.parameter.semestre, e.parameter.grupo); if (!grupoFolder) return errorJSON("Grupo no encontrado"); const files = grupoFolder.getFiles(); let setEstudiantes = new Set(); while (files.hasNext()) { const nombre = files.next().getName(); if (nombre.toLowerCase().includes("retroalimentacion")) { setEstudiantes.add(nombre.split('-Simulacro')[0].replace(/_/g, " ")); } } return ContentService.createTextOutput(JSON.stringify(Array.from(setEstudiantes))).setMimeType(ContentService.MimeType.JSON); } if (action === "datos") { const semestre = e.parameter.semestre; const grupo = e.parameter.grupo; const estudiante = e.parameter.estudiante; const nombreBuscado = limpiar(estudiante); const grupoFolderRetro = getGrupoFolder(rootFolder, semestre, grupo); const filesRetro = grupoFolderRetro.getFiles(); let simulacrosData = []; while (filesRetro.hasNext()) { const f = filesRetro.next(); const nombreArchivo = f.getName(); if (limpiar(nombreArchivo).includes(nombreBuscado) && nombreArchivo.includes("-Simulacro_")) { const match = nombreArchivo.match(/-Simulacro_(\d+)-/); if (match) { simulacrosData.push({ num: parseInt(match[1]), csv: f.getBlob().getDataAsString("UTF-8") }); } } } simulacrosData.sort((a, b) => a.num - b.num); return ContentService.createTextOutput(JSON.stringify({ simulacros: simulacrosData })).setMimeType(ContentService.MimeType.JSON); } return errorJSON("Acción no válida"); } catch (error) { return errorJSON(error.toString()); } } function getFolderByName(parentFolder, name) { const folders = parentFolder.getFolders(); while (folders.hasNext()) { const folder = folders.next(); if (folder.getName().toLowerCase() === name.toLowerCase()) return folder; } return null; } function getGrupoFolder(rootFolder, semestre, grupo) { const semestreFolder = getFolderByName(rootFolder, semestre); if (!semestreFolder) return null; return getFolderByName(semestreFolder, grupo); } function errorJSON(msg) { return ContentService.createTextOutput(JSON.stringify({ error: msg })).setMimeType(ContentService.MimeType.JSON); }
